/ Get the bounding rectangle of the dropDownPane itself
    var paneRect = dropDownPane.BoundingRectangle;

    Console.WriteLine($"Pane X:{paneRect.X} Y:{paneRect.Y} W:{paneRect.Width} H:{paneRect.Height}");
    Console.WriteLine($"Total items: {listItems.Length}, Target index: {targetIndex}");

    // Calculate each item height based on total pane height / number of items
    double itemHeight = (double)paneRect.Height / listItems.Length;

    Console.WriteLine($"Calculated item height: {itemHeight}");

    // Calculate the center X of the pane
    int clickX = (int)(paneRect.X + paneRect.Width / 2);

    // Calculate the center Y of the target item
    int clickY = (int)(paneRect.Y + (itemHeight * targetIndex) + (itemHeight / 2));

    Console.WriteLine($"Clicking at X:{clickX} Y:{clickY}");

    // Use Windows cursor to click at the calculated position
    FlaUI.Core.Input.Mouse.Click(new System.Drawing.Point(clickX, clickY));
